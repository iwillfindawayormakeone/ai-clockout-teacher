One-time setup

Turning the preview into the live website

This page is for the person who sets the site up once. After these steps the board never touches any of it.

1. Give the site its own web address

  1. The association registers the address in its own name, with the board’s email. Not in any one person’s name.
  2. The site’s files are plain web pages. They go on a static host, which is free or close to it for a site this size.
  3. From then on the site can read the association’s Google Sheets, which the preview link is not allowed to do.

2. Switch on email to website

How it works: an email the board marks for the website becomes one row in the News & Events sheet. The website reads that sheet. Nothing else is involved, and the association owns every piece.

  1. Sign in to Google as the board account, the one that receives the HOA email.
  2. Give that account edit access to the News & Events sheet.
  3. Go to script.google.com, start a new project, and paste in the script below.
  4. At the top of the script, check the sheet ID, the forwarding address, and the list of board members allowed to forward.
  5. Choose the function setup and press Run. Google asks for permission once. Say yes.

From then on, every ten minutes, the script picks up any email that was dragged onto the Website label or forwarded to the +website address by a listed board member. Emails from anyone else are set aside under “Website / Not posted” and never reach the site.

/**
 * The Orange's HOA: email to website.
 *
 * Lives in the board's own Google account (the one that receives the HOA email).
 * Every ten minutes it looks for emails the board marked for the website and adds
 * each one as a row in the "News & Events" Google Sheet. The website reads that sheet.
 *
 * Two ways to mark an email:
 *   1. In Gmail, drag the email onto the label "Website" (or open it and choose the label).
 *   2. Forward it to the board address with +website added, for example
 *      orangeshoa+website@gmail.com. Only senders in ALLOWED are accepted this way.
 *
 * One-time setup: paste this file at script.google.com while signed in as the board
 * account, fill in the three settings below, choose the function "setup" and press Run.
 */
const SHEET_ID = '1ZyhU9vLEZCY6NrIuZXuPKBHNfk0fM5mnTzIwWVYEQkw';   // the News & Events sheet (Date, Type, Title, Details)
const ALIAS = 'orangeshoa+website@gmail.com';                       // the forwarding address
const ALLOWED = ['orangeshoa@gmail.com'];                           // board members' own addresses, lower case

const LABEL = 'Website', DONE = 'Website/Posted', SKIPPED = 'Website/Not posted';
const MONTHS = ['jan', 'feb', 'mar', 'apr', 'may', 'jun', 'jul', 'aug', 'sep', 'oct', 'nov', 'dec'];

function setup() {
  [LABEL, DONE, SKIPPED].forEach(function (n) { if (!GmailApp.getUserLabelByName(n)) GmailApp.createLabel(n); });
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'run') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('run').timeBased().everyMinutes(10).create();
  run();
}

function run() {
  const sheet = SpreadsheetApp.openById(SHEET_ID).getSheets()[0];
  const label = GmailApp.getUserLabelByName(LABEL), done = GmailApp.getUserLabelByName(DONE), skipped = GmailApp.getUserLabelByName(SKIPPED);
  const seen = {};
  // 1. dragged onto the label: trusted, because only the board can open this mailbox
  label.getThreads(0, 20).forEach(function (th) { seen[th.getId()] = 1; post(sheet, th); th.removeLabel(label).addLabel(done); });
  // 2. forwarded to the +website address: only from the board members listed in ALLOWED
  GmailApp.search('to:' + ALIAS + ' -label:' + DONE.replace('/', '-') + ' -label:' + SKIPPED.replace(/[ \/]/g, '-') + ' newer_than:7d', 0, 20).forEach(function (th) {
    if (seen[th.getId()]) return;
    const msgs = th.getMessages(), from = (msgs[msgs.length - 1].getFrom().match(/[\w.+-]+@[\w.-]+/) || [''])[0].toLowerCase();
    if (ALLOWED.indexOf(from) > -1) { post(sheet, th); th.addLabel(done); } else { th.addLabel(skipped); }
  });
}

function post(sheet, thread) {
  const msgs = thread.getMessages(), m = msgs[msgs.length - 1];
  const p = readEmail('Subject: ' + m.getSubject() + '\n\n' + m.getPlainBody());
  sheet.appendRow([new Date(), 'News', p.title, p.text]);
  // A date in the subject line ("Board meeting December 9") also puts it on the calendar.
  if (p.when) sheet.appendRow([p.when, 'Event', p.title, p.text]);
}

/** Turns a forwarded email into a headline and a short paragraph. Same rules as the website's own preview. */
function readEmail(raw) {
  let title = '';
  let lines = String(raw).replace(/\r\n?/g, '\n').split('\n').map(function (l) { return l.replace(/^>+\s?/, '').trim(); });
  lines = lines.filter(function (l) {
    const s = /^Subject:\s*(.+)$/i.exec(l); if (s) { if (!title) title = s[1]; return false; }
    return !/^(-{3,}.*(forwarded|original) message.*-{3,}|begin forwarded message:|(from|to|cc|bcc|date|sent|reply-to):.*)$/i.test(l);
  });
  while (lines.length && !lines[0]) lines.shift();
  title = (title || 'A note from the board').replace(/^((fwd?|fw|re)\s*:\s*)+/i, '').trim().slice(0, 110);
  let end = -1;
  lines.forEach(function (l, i) { if (end < 0 && i > 0 && /^(--|sent from my .*|thanks[,!.]?|thank you[,!.]?|best[,.]?|regards[,.]?|sincerely[,.]?)$/i.test(l)) end = i; });
  if (end > 0) lines = lines.slice(0, end);
  const text = lines.join('\n').replace(/^(hi|hello|dear)\b[^\n]{0,40}\n+/i, '').trim().replace(/\s*\n\s*/g, ' ').slice(0, 700);
  let when = null;
  const d = /\b(January|February|March|April|May|June|July|August|September|October|November|December|Jan|Feb|Mar|Apr|Jun|Jul|Aug|Sept|Sep|Oct|Nov|Dec)\.?\s+(\d{1,2})(?:st|nd|rd|th)?(?:,?\s+(\d{4}))?/i.exec(title);
  if (d) {
    const today = new Date(); today.setHours(0, 0, 0, 0);
    const day = new Date(d[3] ? +d[3] : today.getFullYear(), MONTHS.indexOf(d[1].slice(0, 3).toLowerCase()), +d[2]);
    if (!d[3] && day < today) day.setFullYear(day.getFullYear() + 1);
    if (day >= today) when = day;
  }
  return { title: title, text: text, when: when };
}

3. Later, if the board wants them

  • Resident accounts (My home): the database and the sign-in are built and tested with sample homes. Switching them on means loading the real addresses.
  • Reminders by email or text: needs a sending service. Email is free at this size. Text messages cost a little per message.
  • A tidier rewrite of long emails: an AI step can shorten a long email into two sentences before it posts. Optional, and it costs a few cents a month.
